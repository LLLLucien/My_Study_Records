# C++ 智能指针

## 智能指针是什么

-简历上写熟悉STL，智能指针，lambda表达式

- 智能指针就是相当于用类对象包装一个裸指针（T*），让它替程序员自动管理内存。用法和裸指针几乎一样（->、* 都支持），但不用手动 delete

## 简单回顾

- STL
  - 容器
    - 顺序容器
    - 有序关联容器
    - 无序关联容器
    - 容器适配器
  - 算法
  - 迭代器

## 传统指针缺点

- 传统裸指针缺点需要手动释放，容易忘记

## 核心思想

- RAII 资源获取即初始化，将堆内存交给类对象管理
- 生命周期结束（析构）时，自动释放内存

## 三种智能指针类型

- unique_ptr 独占所有权
- shared_ptr 共享所有权
- weak_ptr

## 基本使用

```c++
    //创建方法
    unique_ptr<Person> p1(new Person("Tom"));
    make_unique<Person>("Jerry");
    // 得到裸指针
    p1.get();
    // 返回裸指针，并释放p1的所有权，不是delete
    p1.release();
    // 释放p1的所有权，并将p1置空
    p1.reset();
    // 释放p1的所有权，并将p1置空，可以传参数，使p1指向一个新的对象
    p1.reset(new Person("Tom"));
    // 交换指针的方法
    p1.swap(p2);
    swap(p1, p2);
    // 查看引用计数，独占指针没有引用计数
    // 主要是共享指针和弱指针有
    p1.use_count();


```

## 具体实现

### 独占指针

```c++
#include<iostream>
#include<string>

using namespaces std;

class Person
{
    public:
    string name;
    Person(string n):name(n){cout<<"Person (string)"<<endl;}
    ~Person(){cout<<"~Person()"<<endl}
};

int main()
{
    // 智能指针
    // 1. 独占指针的创建
    unique_ptr<Person> p1(new Person("Tom"));
    cout << p1->name << endl;
    // 其他创建
    auto p2 = make_unique<Person>("Jerry");

    // 独占不能直接拷贝，会报错
    // unique_ptr<Person> p2=p1;

    // 2. 独占指针的移动
    unique_ptr<Person> p2 = move(p1);
    cout << p2->name << endl;
    if (p1 == nullptr)
    {
        cout << "p1 is nullptr" << endl;
    }


    // 3.获得p2的裸指针，不要进行delete
    // p2还在
    Person *p3 = p2.get();

    // 4.释放p2的裸指针，p4 指向p2的裸指针
    Person *p4 = p2.release();
    // 这时需要自己delete
    delete p4;

    // 5. 重置指针，相当于手动delete智能指针
    // 可以传参数，使p2指向一个新的对象
    p2.reset();

    // 6. 交换指针
    unique_ptr<Person> p5(new Person("Tom"));
    unique_ptr<Person> p6(new Person("Jerry"));
    p5.swap(p6);  // 指针自带的函数
    swap(p5, p6); // 标准库的函数
    cout << p5->name << endl;
    cout << p6->name << endl;

}

```

### 共享指针

```c++
    // shared.cxx
// 智能指针
#include <iostream>
#include <memory>
#include <string>

using namespace std;

class Person
{
  public:
    string name;
    Person(string n) : name(n)
    {
        cout << "Person (string)" << endl;
    }
    ~Person()
    {
        cout << "~Person()" << endl;
    }
};
void x(shared_ptr<Person> p)
{
    cout << "X: " << p->name << endl;
    cout << "X count: " << p.use_count() << endl;
}
void y(shared_ptr<Person> p)
{
    cout << "Y: " << p->name << endl;
    cout << "Y count: " << p.use_count() << endl;
}

int main()
{
    // 1.创建一个共享指针
    shared_ptr<Person> p1(new Person("Tom"));

    // 可以查看引用计数
    cout << "use_count() = " << p1.use_count() << endl;

    // 2. 共享指针的拷贝
    shared_ptr<Person> p2 = p1;
    shared_ptr<Person> p3 = p2;
    cout << "p1.use_count() = " << p1.use_count() << endl;
    cout << "p2.use_count() = " << p2.use_count() << endl;
    cout << "p3.use_count() = " << p3.use_count() << endl;

    // reset后，count就变成了1，reset是减去1，不是析构
    // 区分手动释放和重置
    // Person *p4 = p2.get();
    // delete p4; // 不可以，会报错

    p2.reset();
    cout << "p1.use_count() = " << p1.use_count() << endl;
    cout << "p2.use_count() = " << p2.use_count() << endl;
    cout << "p3.use_count() = " << p3.use_count() << endl;

    x(p1); // x里面会是count=2，因为函数内部相当于有一个指针，外部有一个指针
    y(p1); // 到达y的时候，x里面的会自动释放，所有y还是保持count=2
    return 0;
}
```

#### 共享指针的循环引用问题

```c++
#include <iostream>
#include <memory>
using namespace std;

struct A;
struct B;

struct A
{
    shared_ptr<B> b;                       // A 持有 B
    ~A() { cout << "~A()" << endl; }
};

struct B
{
    shared_ptr<A> a;                       // B 持有 A，互相指向形成环
    ~B() { cout << "~B()" << endl; }
};

int main()
{
    shared_ptr<A> pa(new A);
    shared_ptr<B> pb(new B);
    pa->b = pb;                            // pb 计数 1 → 2
    pb->a = pa;                            // pa 计数 1 → 2
    cout << "pa: " << pa.use_count() << endl;   // 2
    cout << "pb: " << pb.use_count() << endl;   // 2
    return 0;
    // main 结束时 pa、pb 各减 1，但计数仍为 1
    // 永远不会触发析构（看不到 ~A() / ~B()），内存泄漏
}
```

问题：两个对象互相持有 `shared_ptr`，引用计数始终到不了 0，谁也无法释放。

解决：把其中一个改成 `weak_ptr`（弱引用，不增加计数），如 `struct B { weak_ptr<A> a; };`，环就断了。

### 弱指针

- 目的：解决循环引用问题
- 不会增加引用计数，不会导致引用计数不为 0 时无法释放

```c++
// weak.cxx
#include <iostream>
#include <memory>

using namespace std;

struct Node
{
    // 成员属性默认是public的

    // 成员属性
    int val;
    // 以前裸指针写法
    // struct Node *next;
    weak_ptr<Node> next;
    // shared_ptr<Node> next;

    Node(int val) : val(val)
    {
        cout << "Node(" << val << ")" << endl;
    }
    ~Node()
    {
        cout << "~Node(" << val << ")" << endl;
    }
};

int main()
{
    auto n1 = make_shared<Node>(5);
    shared_ptr<Node> n2(new Node(6));
    // cout << n1.use_count() << endl; // count=1
    // cout << n2.use_count() << endl;
    // n1->next = n2;
    // n2->next = n1;

    // cout << n1.use_count() << endl; // count=2，而且无法释放
    // cout << n2.use_count() << endl;
    // n1.reset();
    // n2.reset();
    // cout << n1.use_count() << endl;
    // cout << n2.use_count() << endl;

    weak_ptr<Node> n3 = n1;
    cout << n1->val << endl;
    // 弱指针不能解引用
    // cout << n3->val << endl;
    // 需要lock()变为共享指针
    cout << n3.lock()->val << endl;
    cout << n3.use_count() << endl;
    return 0;
}
```
